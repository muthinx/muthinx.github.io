	OPERATOR
	simbol yang digunakan untuk melakukan operasi 
	aritmatika, penugasan, perbandingan, logika, string = operator binary
	kondisional = operator ternary
	typeof = operator unary
	OPERATOR ARITMATIKA
	+ - * / % . % operator modulo / modulus / sisa bagi 5%2 = 1
	urutan operasi aritmatika KuKaBaTaKu ()*/+-
	
	OPERATOR PENUGASAN
	var x
	x=10 -- setelah ini x enter muncul 10
	= += -= *= /= %= ini juga operator penugasan
	
	OPERATOR PERBANDINGAN
	== != === !== > < >= <= operator ini menghasilkan nilai boolean
	10 == 10 hasilnya true 10 == 15 hasilnya false
	10 == "10" hasilnya true 10 === "10" hasilnya false
	
	OPERATOR LOGIKA
	&& untuk AND || !
	var x = 5
	(x % 2 ==0) && (x > 10)
	|| untuk OR
	! untuk NOT fungsinya membalikkan hasil boolean

	OPERATOR STRING
	+
	var a="apple" var b="pen"
	a + b enter "applepen" a + " " + b = "apple pen"
	10 + "10" hasilnya "1010"
	10 + 10 + "10" hasilnya "2010"
	N+N=N N+S=S

	OPERATOR TYPEOF
	typeof(10) hasilnya number
	typeof("10") hasilnya string

	OPERATOR KONDISIONAL
	(kondisi) ? benar : salah
	(x % 2 ==) ? "genap" : "ganjil"
